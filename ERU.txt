function searchExams() {
    const idVal = parseInt(document.getElementById('studentId').value);
    const dept = document.getElementById('deptSelect').value;

    if (!idVal) {
        alert("يرجى كتابة رقم القيد أولاً!");
        return;
    }

    const results = db.filter(item => {
        const inRange = (idVal >= item.from && idVal <= item.to) || item.from === "ALL";
        const inDept = (dept === "ALL") || (item.dept === dept);
        return inRange && inDept;
    });

    displayResults(results, idVal);
}

function displayResults(items, id) {
    const tableBody = document.getElementById('examTableBody');
    const resultSec = document.getElementById('resultSection');
    const countdownCard = document.getElementById('countdownCard');
    document.getElementById('dispId').innerText = id;

    tableBody.innerHTML = '';

    if (items.length === 0) {
        tableBody.innerHTML = `<tr><td colspan="4" style="text-align:center; padding:30px; color:var(--text-muted);">لم يتم العثور على امتحانات مسجلة لرقم القيد هذا.</td></tr>`;
    } else {
        items.forEach(item => {
            tableBody.innerHTML += `
                <tr class="exam-row">
                    <td>
                        <span class="course-code">${item.code}</span>
                        <div class="course-name">${item.title}</div>
                    </td>
                    <td><strong>${item.day}</strong><br><small style="color:var(--text-muted);">${item.date}</small></td>
                    <td><i class="fa-regular fa-clock" style="color:var(--gold);"></i> ${item.time}</td>
                    <td><span class="building-tag">${item.bldg}</span><br><small>قاعة: ${item.hall}</small></td>
                </tr>
            `;
        });
        setupCountdown(items[0].date);
        countdownCard.style.display = 'block';
    }

    resultSec.style.display = 'block';
}

function setupCountdown(examDateStr) {
    const parts = examDateStr.split('/');
    const targetDate = new Date(parts[2], parts[1] - 1, parts[0], 9, 30);
    const now = new Date();
    const diff = targetDate - now;

    if (diff > 0) {
        const days = Math.floor(diff / (1000 * 60 * 60 * 24));
        const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
        const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));

        document.getElementById('tDays').innerText = days;
        document.getElementById('tHours').innerText = hours;
        document.getElementById('tMins').innerText = mins;
    }
}
